## B10: Recursive Node Discovery

### Objective
Update `Graph__Repository` to find ALL `issue.json` files recursively, including nested `issues/` subfolders.

### Problem
Current code only searches `data/{type}/{label}/` — misses issues at:
```
data/project/Project-1/issues/Version-1/issue.json
data/project/Project-1/issues/Version-1/issues/Task-6/issue.json
```

### Implementation

**File:** `mgraph_ai_ui_html_transformation_workbench/service/issues/graph_services/Graph__Repository.py`

**Add dataclass for node info:**
```python
from dataclasses import dataclass

@dataclass
class Node__Info:
    label     : str                                                              # e.g., "Task-6"
    path      : str                                                              # e.g., "data/project/Project-1/issues/Version-1/issues/Task-6"
    node_type : str                                                              # e.g., "task"
```

**Add new method:**
```python
# ═══════════════════════════════════════════════════════════════════════════════
# Node Discovery (Phase 2: Recursive Search)
# ═══════════════════════════════════════════════════════════════════════════════

def nodes_list_all(self, root_path: str = '') -> List[Node__Info]:               # Find all issues recursively
    """Find all issue.json files, including nested issues/ folders.
    
    Args:
        root_path: If specified, only return nodes under this path.
                   Empty string means search everything.
    
    Returns:
        List of Node__Info with label, path, and node_type.
    """
    all_paths   = self.storage_fs.files__paths()
    nodes       = []
    skip_labels = {'config', 'data', 'issues', 'indexes', '.issues'}             # System folders
    
    for path in all_paths:
        if path.endswith('/issue.json') is False:                                # Only issue.json files
            continue
        
        if root_path:                                                            # Filter by root if specified
            if self._is_under_path(path, root_path) is False:
                continue
        
        folder_path = path.rsplit('/issue.json', 1)[0]                           # Extract folder and label
        label       = folder_path.rsplit('/', 1)[-1]
        
        if label in skip_labels:                                                 # Skip system folders
            continue
        
        node_type = self._extract_node_type_from_path(path)                      # Get type from file
        
        nodes.append(Node__Info(label=label, path=folder_path, node_type=node_type))
    
    return nodes

def _is_under_path(self, file_path: str, root_path: str) -> bool:                # Check path containment
    """Check if file_path is under root_path."""
    if not root_path:
        return True
    if root_path.endswith('/'):
        return file_path.startswith(root_path)
    return file_path.startswith(f"{root_path}/") or file_path.startswith(f"{root_path}/issues/")

def _extract_node_type_from_path(self, file_path: str) -> str:                   # Get node_type from JSON
    """Extract node_type from issue.json content."""
    try:
        content = self.storage_fs.file__str(file_path)
        if content:
            data = json_loads(content)
            return data.get('node_type', '')
    except:
        pass
    return ''
```

def node_load_by_path(self, folder_path: str) -> Schema__Node:                   # Load by explicit path
    """Load node from path like 'data/project/Project-1/issues/Version-1'."""
    issue_file = f"{folder_path}/issue.json"
    
    if self.storage_fs.file__exists(issue_file) is False:
        return None

def node_find_path_by_label(self, label: str) -> str:                            # Find path for label
    """Find folder path for a label by searching all paths."""
    all_paths = self.storage_fs.files__paths()
    
    for path in all_paths:
        if path.endswith(f'/{label}/issue.json'):
            return path.rsplit('/issue.json', 1)[0]
    
    return None

def _delete_legacy_node_json(self, node_type: str, label: str) -> bool:          # Remove legacy file
    """Delete node.json if exists alongside issue.json."""
    if self.path_handler.has_base_path():
        node_path = f"{self.path_handler.base_path}/data/{node_type}/{label}/node.json"
    else:
        node_path = f"data/{node_type}/{label}/node.json"
    
    if self.storage_fs.file__exists(node_path):
        return self.storage_fs.file__delete(node_path)
    
    return False

### Objective

### Prerequisites

### Implementation

**Migration script (run before deploying B13):**
```python
def migrate_all_to_issue_json(repository: Graph__Repository) -> dict:
    """Convert all remaining node.json files to issue.json."""
    all_paths = repository.storage_fs.files__paths()
    results   = {'converted': 0, 'skipped': 0, 'deleted': 0}
    
    for path in all_paths:
        if not path.endswith('/node.json'):
            continue
        
        issue_path = path.replace('/node.json', '/issue.json')
        
        if repository.storage_fs.file__exists(issue_path):
            # issue.json exists, just delete node.json
            repository.storage_fs.file__delete(path)
            results['deleted'] += 1
        else:
            # Copy node.json → issue.json, then delete
            content = repository.storage_fs.file__bytes(path)
            repository.storage_fs.file__save(issue_path, content)
            repository.storage_fs.file__delete(path)
            results['converted'] += 1
    
    return results
```

## B14: Path-Based Node Endpoint

### Objective
Add endpoint to load node by hierarchical path (e.g., `Project-1/Version-1/Task-6`).

### Endpoint
```
GET /nodes/api/node/by-path?path={hierarchical_path}
```

### Implementation

**File:** `Node__Service.py`

```python
def resolve_hierarchical_path(self, path: str, current_root: str = '') -> str:   # Resolve to filesystem path
    """Convert 'Project-1/Version-1/Task-6' to filesystem path."""
    if not path:
        return current_root or None
    
    segments = path.split('/')
    
    if current_root:
        result    = current_root                                                 # Start from root
        start_idx = 0
    else:
        first_label = segments[0]                                                # Find first segment
        result      = self.repository.node_find_path_by_label(first_label)
        
        if not result:
            return None
        
        start_idx = 1
    
    for segment in segments[start_idx:]:                                         # Each segment in issues/
        result = f"{result}/issues/{segment}"
    
    return result

def get_node_by_hierarchical_path(self, path: str) -> Schema__Node__Response:    # Get by hierarchical path
    """Resolve path and load node."""
    current_root = ''
    if self.root_selection_service:
        current_root = str(self.root_selection_service.current_root)
    
    resolved = self.resolve_hierarchical_path(path, current_root)
    
    if not resolved:
        return Schema__Node__Response(success=False, message=f'Cannot resolve: {path}')
    
    node = self.repository.node_load_by_path(resolved)
    
    if node:
        return Schema__Node__Response(success=True, node=node)
    
    return Schema__Node__Response(success=False, message=f'Not found at: {resolved}')
```

**File:** `Routes__Nodes.py`

```python
@route_path('/nodes/api/node/by-path')

            return {'success': True, 'node': node.json()}
        return {'success': False, 'message': f'Not found: {label}'}
    
    @route_path('/graph/api/node/{label}/children')
    def get_children(self, label: str):
        graph = self.sync_service.ensure_loaded()
        node  = graph.get_node_by_label(label)
        if node:
            children = graph.get_children(str(node.node_id))
            return {'success': True, 'nodes': [c.json() for c in children]}
        return {'success': False, 'message': f'Not found: {label}'}
```

## B22: Hyphenated Label Support

### Objective
Generate labels like `User-Story-1` instead of `UserStory1`.

### Implementation

**Update label generation:**
```python
def generate_label(self, node_type: str, index: int) -> str:                     # Hyphenated labels
    """Generate 'User-Story-1' from type 'user-story' and index 1."""
    words        = node_type.split('-')
    display_type = '-'.join(word.capitalize() for word in words)
    return f"{display_type}-{index}"
```

**Examples:**
| Type | Index | Label |
|------|-------|-------|
| `task` | 1 | `Task-1` |
| `user-story` | 5 | `User-Story-5` |
| `git-repo` | 1 | `Git-Repo-1` |

**Backward compatible:** Folder search already works for both formats.
